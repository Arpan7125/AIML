feature_encoder = OrdinalEncoder()

11. Compare Performance with Decision Tree and Logistic Regression¶

In [ ]:
# Train three models and compare them briefly
models = {
    'Categorical Naive Bayes': CategoricalNB(),
    'Decision Tree': DecisionTreeClassifier(random_state=42),
    'Logistic Regression': LogisticRegression(random_state=42, max_iter=1000)
}

results = {}

print("Training models...")
for name, model in models.items():
    model.fit(X_train, y_train)
    y_pred = model.predict(X_test)
    acc = accuracy_score(y_test, y_pred)
    results[name] = {
        'model': model,
        'predictions': y_pred,
        'accuracy': acc
    }
    print(f"- {name}: {acc:.2f}")

print("Done. The table below shows the simple comparison.")

Categorical Naive Bayes: Accuracy = 0.6000 (60.00%)
Decision Tree: Accuracy = 0.4000 (40.00%)
